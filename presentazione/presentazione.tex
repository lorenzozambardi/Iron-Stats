\begin{frame}{1.1 Limiti dei Modelli Discreti Tradizionali}
    \begin{itemize}
        \item \textbf{Approccio Scalare Classico:}
        \begin{itemize}
            \item $\text{Volume} = \text{Serie} \times \text{Ripetizioni}$
            \item $\text{Tonnellaggio grezzo} = \text{Serie} \times \text{Reps} \times \text{Carico}$
        \end{itemize}
        \vspace{0.3cm}
        \item \textbf{Criticità Fisiologiche e Meccaniche:}
        \begin{itemize}
            \item Incapacità di modellare la variazione continua del \textbf{braccio di leva} muscolo-articolare.
            \item Ignora il \textbf{profilo di resistenza} delle macchine o dei pesi liberi lungo il Range of Motion (ROM).
            \item Mancata stima dell'\textbf{affaticamento neuromuscolare intrinseco} e della caduta di velocità concentrica.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{1.2 Formulazione del Problema Numerico}
    \begin{block}{Transizione al Continuo}
        Si passa da variabili scalari a \textbf{campi funzionali continui} parametrizzati sull'angolo o quota di escursione articolare:
        $$t \in [0, 1] \quad (\text{Range of Motion - ROM})$$
    \end{block}
    \vspace{0.2cm}
        L'ottimizzazione del programma settimanale viene formulata come un \textbf{Problema di minimo globale multiobiettivo vincolato su spazio discreto combinatorio}
\end{frame}

\begin{frame}{1.3 Obiettivi di Calcolo Scientifico e Metodi Numerici}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Approssimazione Funzionale}
                Studio comparativo di accuratezza, stabilità e condizionamento numerico:
                \begin{itemize}
                    \item Interpolazione di Lagrange/Newton
                    \item Spline Cubiche ($C^2$)
                    \item Curve Parametriche di Bézier (Bernstein)
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Quadratura \& Ottimizzazione}
                \begin{itemize}
                    \item Conservazione rigorosa della magnitudo tramite \textbf{formula composita dei trapezi}.
                    \item Adimensionalizzazione dei gradienti d'errore via simulazione \textbf{Monte Carlo}.
                    \item Euristica di convergenza globale: \textbf{Simulated Annealing}.
                \end{itemize}
            \end{block}

\end{frame}

\begin{frame}{2.1 Formalizzazione della Classe Esercizio}
    \begin{block}{Parametri Biomeccanici}
        \begin{itemize}
            \item $c_{load} \in (0, 1]$: Efficienza articolare e dispersione del carico.
            \item $f_{base} \in [1, 10]$: Coefficiente intrinseco di affaticamento sistemico.
            \item $\mu_{load} \in \mathbb{R}^+, \; q_{load} \in \mathbb{R}$: Correzione carico effettivo: $L = L_{nom} \cdot \mu_{load} + q_{load}$.

        \end{itemize}
    \end{block}
    \begin{alertblock}{Vincolo di Conservazione della Ripartizione Muscolare}

        $$\sum_{m \in \mathcal{M}} M_m = 1.0 \pm \varepsilon, \quad \varepsilon \le 10^{-4}$$
        \footnotesize \textbf{Tolleranza numerica $\varepsilon$:} dovuta all'aritmetica floating-point IEEE 754 e all'arrotondamento delle percentuali nei file di configurazione JSON ($M_m = \text{pct} / 100$).
    \end{alertblock}
\end{frame}

% METODOLOGIA 1: LAGRANGE
% -------------------------------------------------------------
\begin{frame}{2.2 Metodo 1: Interpolazione Polinomiale di Lagrange (Teoria)}
    \begin{block}{Formulazione Globale di Lagrange}
        Dati $n$ punti campionati $(t_i, y_i)$ lungo il ROM $t \in [0, 1]$:
        $$p_{n-1}(t) = \sum_{i=1}^n y_i \ell_i(t), \qquad \ell_i(t) = \prod_{j \ne i} \frac{t - t_j}{t_i - t_j}$$
    \end{block}
    \vspace{0.15cm}
    \begin{alertblock}{Condizionamento e Fenomeno di Runge}
        \begin{itemize}
            \item \textbf{Matrice di Vandermonde:} $V \mathbf{a} = \mathbf{y}$ con $V_{ij} = t_i^{j-1}$. Il numero di condizionamento cresce esponenzialmente: $\kappa(V) \sim \mathcal{O}(e^{\gamma n}) \implies$ perdita totale di stabilità numerica.
            \item \textbf{Fenomeno di Runge (Nodi Equispaziati):}
            $$\lim_{n \to \infty} \max_{t \in [0, 1]} |f(t) - p_{n-1}(t)| = \infty$$
            L'errore diverge esponenzialmente in corrispondenza degli estremi ($t \to 0$ e $t \to 1$).
        \end{itemize}
    \end{alertblock}
\end{frame}

\begin{frame}{2.2 Metodo 1: Lagrange su Nodi Equispaziati (Caso LR)}
    \centering
    \includegraphics[height=0.54\textheight, keepaspectratio]{plot_lagrange_runge.png}
    \begin{exampleblock}{Criticità Biomeccanica Evidenziata}
        \footnotesize Con soli $n=9$ nodi equispaziati, l'interpolante diverge drasticamente nella fase di allungamento iniziale ($t < 0.2$) e di accorciamento finale ($t > 0.85$), generando picchi di tensione artificiali e del tutto non fisiologici.
    \end{exampleblock}
\end{frame}

\begin{frame}{2.2 Metodo 1: Nodi di Chebyshev-Gauss-Lobatto (Caso LR)}
    \centering
    \includegraphics[height=0.48\textheight, keepaspectratio]{plot_lagrange_gauss_lobatto.png}
    \begin{alertblock}{Perché Gauss-Lobatto non risolve il problema?}
        \footnotesize \textbf{Distribuzione CGL:} $t_k = \frac{1}{2}\left(1 - \cos\frac{k\pi}{N-1}\right)$ addensa i nodi agli estremi ed elimina Runge teorico, ma:
        \begin{itemize}
            \item Servono \textbf{troppi parametri} ($N \ge 12 \div 16$) per tentare di catturare la zona piatta di quiete ($t \in [0, 0.2]$).
            \item Permangono \textbf{ondulazioni parassite} e valori negativi $y(t) < 0$, rendendo il modello instabile e non parsimonioso.
        \end{itemize}
    \end{alertblock}
\end{frame}

% METODOLOGIA 2: SPLINE CUBICHE
% -------------------------------------------------------------
\begin{frame}{2.2 Metodo 2: Spline Cubiche Interpolanti $C^2$ (Teoria)}
    \begin{block}{Definizione e Risoluzione Lineare Tridiagonale}
        Dato l'insieme di nodi $0 = t_0 < t_1 < \dots < t_n = 1$, la spline cubica è un polinomio di grado 3 su ciascun sottointervallo $I_k = [t_k, t_{k+1}]$ con continuità di classe $C^2$:
        $$S(t) \in C^2[0, 1] \implies S_k(t_k) = y_k, \quad S'_k(t_k) = S'_{k-1}(t_k), \quad S''_k(t_k) = S''_{k-1}(t_k)$$
    \end{block}
    \vspace{0.15cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Vantaggi Computazionali}
                \begin{itemize}
                    \item Continuità di derivata prima e curvatura.
                    \item Risoluzione del sistema tridiagonale in $\mathcal{O}(n)$ con \textbf{Algoritmo di Thomas}.
                    \item Elimina le oscillazioni globali di Runge.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{Limite Fisico Fondamentale}
                \begin{itemize}
                    \item \textbf{Assenza di forma vincolata:} la spline cubica non rispetta la monotonicità locale né la positività.
                    \item Tende a produrre \textbf{undershoot non fisici}:
                    $$y(t) < 0$$
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2.2 Metodo 2: Spline Cubica Naturale su LR (Undershoot)}
    \centering
    \includegraphics[height=0.52\textheight, keepaspectratio]{plot_spline_overshoot.png}
    \begin{alertblock}{Violazione dei Principi Biomeccanici}
        \footnotesize L'interpolazione spline naturale sui nodi fisiologici genera un marcato undershoot ($y(t) < 0$) per $t \in [0, 0.2]$. Un muscolo scheletrico non può esercitare tensione attiva negativa (repulsiva); richiederebbe forzature numeriche ad-hoc (clamping artificiale).
    \end{alertblock}
\end{frame}

\begin{frame}{2.2 Metodo 2: Densità dei Nodi e Dilemma delle Spline}
    \centering
    \includegraphics[height=0.52\textheight, keepaspectratio]{plot_spline_densita.png}

    \end{exampleblock}
\end{frame}

\begin{frame}{2.2 Metodo 3: Curve di Bézier e Polinomi di Bernstein (Teoria)}
    \begin{block}{Definizione Vettoriale Bidimensionale ($x, y$)}
        Ogni punto ha due coordinate: $\mathbf{P}_i = \begin{pmatrix} x_i \\ y_i \end{pmatrix} \implies \mathbf{P}(t) = \begin{pmatrix} x(t) \\ y(t) \end{pmatrix} = \sum_{i=0}^3 \begin{pmatrix} x_i \\ y_i \end{pmatrix} B_{i,3}(t)$
    \end{block}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Invarianza Affine}
                Poiché i pesi sommano a uno ($\sum_{i=0}^3 B_{i,3} \equiv 1$):
                $$T\big(\mathbf{P}(t)\big) = \sum_{i=0}^3 T(\mathbf{P}_i) B_{i,3}(t)$$
                \footnotesize Traslazioni, rotazioni e \textbf{cambi di scala} (es. da gradi a $[0, 1]$ o carichi in kg) trasformano i soli 4 punti $\mathbf{P}_i$ senza dover ricalcolare la curva.
            \end{exampleblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{Inviluppo Convesso (Convex Hull)}
                Poiché i pesi sono non-negativi ($B_{i,3} \ge 0$):
                $$\mathbf{P}(t) \in \text{Conv}(\{\mathbf{P}_0, \mathbf{P}_1, \mathbf{P}_2, \mathbf{P}_3\})$$
                \footnotesize \textbf{Garanzia fisica:}
                $$y_i \ge 0 \;\; \forall i \implies y(t) \ge 0 \;\; \forall t$$
                Tensione muscolare sempre non-negativa per costruzione geometrica.
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2.2 Metodo 3: Come Funziona Geometricamente (De Casteljau)}
    \begin{block}{Interpolazioni Lineari Annidate (LERP Ricorsivo a 3 Livelli)}
        Fissato $t \in [0, 1]$, il punto sulla curva $\mathbf{P}(t)$ si ottiene ricorsivamente:
        \begin{itemize}
            \item \textbf{Livello 1:} $Q_i(t) = (1-t)P_i + t P_{i+1}, \quad i \in \{0, 1, 2\}$
            \item \textbf{Livello 2:} $R_i(t) = (1-t)Q_i(t) + t Q_{i+1}(t), \quad i \in \{0, 1\}$
            \item \textbf{Livello 3:} $\mathbf{P}(t) = (1-t)R_0(t) + t R_1(t)$
        \end{itemize}
    \end{block}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \begin{exampleblock}{Ruolo Fisico dei Punti di Controllo}
                \begin{itemize}
                    \item $P_0, P_3$: \textbf{ancore di attivazione} ($t=0, 1$ locale). Delimitano $[x_0, x_3]$ e \textbf{non sono vincolati} a $x=0, 1$ (es. LR: $x_0=0.2, x_3=0.9$, con $y=0$ altrove).
                    \item $P_1, P_2$: \textbf{attrattori tangenziali} della curvatura.
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.46\textwidth}
            \begin{block}{Stabilità \& Dead Zones}
                \begin{itemize}
                    \item Sole combinazioni convesse ($t \in [0, 1]$) $\implies$ totale stabilità numerica.
                    \item Modella naturalmente escursioni con contrazione parziale nel ROM.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

    \centering
    \includegraphics[height=0.52\textheight, keepaspectratio]{plot_de_casteljau.png}
    \begin{exampleblock}{Visualizzazione Dinamica della Costruzione}

\begin{frame}{2.2 Metodo 3: Come si Calcola la Tensione $y(x)$ nel Software}
    \begin{block}{Il Problema dell'Inversione Cinematica}
        Bézier è parametrizzata in $t \in [0, 1] \implies \begin{cases} x(t) = (1-t)^3 x_0 + 3t(1-t)^2 x_1 + 3t^2(1-t) x_2 + t^3 x_3 \\ y(t) = (1-t)^3 y_0 + 3t(1-t)^2 y_1 + 3t^2(1-t) y_2 + t^3 y_3 \end{cases}$ \\
        Per calcolare la tensione $y$ dato un angolo di ROM target $x_{target} \in [0, 1]$:
    \end{block}
    \vspace{0.1cm}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \begin{alertblock}{Risoluzione Radice $x(t) - x_{target} = 0$}
                \begin{itemize}
                    \item Poiché $x_0 < x_1 < x_2 < x_3$, la funzione $x(t)$ è \textbf{strettamente monotona crescente}.
                    \item Esiste un'\textbf{unica soluzione} $t^* \in [0, 1]$.
                    \item Trovata con \textbf{bisezione} in max 15 iterazioni:
                    $$|x(t^{(k)}) - x_{target}| < 10^{-3}$$
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Valutazione Diretta di $y$}
                Una volta calcolato $t^*$, la tensione vale:
                $$y_{target} = y(t^*)$$
                \begin{itemize}
                    \item Se $x_{target} < x_0$ o $x_{target} > x_3 \implies y = 0$.
                    \item Costo costante $\mathcal{O}(1)$ in frazioni di microsecondo.
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2.3 Quadratura Numerica e Riscalamento Vincolato}
    \begin{block}{Calcolo dell'Area Sotto la Curva (AUC)}
        Griglia discreta $N = 50$, passo $h = \Delta t = \frac{1}{N-1}$:
        $$I_{trap} = \int_0^1 y(x)\,dx \approx \sum_{k=0}^{N-2} \frac{y(t_k) + y(t_{k+1})}{2} (t_{k+1} - t_k)$$
        Errore di troncamento della formula composita: $E_{trap} = -\frac{1}{12(N-1)^2} y''(\xi) = \mathcal{O}(h^2)$.
    \end{block}
    \begin{block}{Operatore di Riscalamento Funzionale}
        Garantisce che l'integrale coincida esattamente con la magnitudo target $M$:
        $$\tilde{y}(t) = \begin{cases} y(t) \cdot \frac{M}{I_{trap}} & \text{se } I_{trap} > 0 \\ \mathbf{0} & \text{altrimenti} \end{cases} \implies \int_0^1 \tilde{y}(t)\,dt = M + \mathcal{O}(h^2)$$
    \end{block}
\end{frame}

\section{3. Analizzatore: Dinamica Discreta e Metriche di Carico}
% =============================================================

\begin{frame}{3.1 Volume ($V$) ed Effective Volume ($V_{eff}$)}
    \begin{itemize}
        \item \textbf{Ripetizioni Totali Equivalenti (serie $s$):}
        $$R_{tot, s} = R_{base, s} + c_{ass} R_{ass, s} + c_{part} R_{part, s} \quad (c_{ass}=0.5, \, c_{part}=0.33)$$
        \item \textbf{Ripetizioni Efficaci da Scala RPE ($RPE \in [0, 10]$):}
        $$R_{eff, s} = \min\Big(R_{base, s}, \, \max\big(0, \, RPE_s - 4.0\big)\Big) + c_{ass} R_{ass, s} + c_{part} R_{part, s}$$
        \item \textbf{Proiezione sulla quota muscolare $\Omega_m = \int \tilde{y}_m(t) dt$:}
        $$V = \sum_{s} R_{tot, s} \cdot \Omega_m, \qquad V_{eff} = \sum_{s} R_{eff, s} \cdot \Omega_m$$
    \end{itemize}
\end{frame}

\begin{frame}{3.2-3.3 Tonnellaggio e Tonnellaggio Ponderato}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Tonnellaggio ($T$ e $T_{eff}$)}
                Carico effettivo: $L_s = L_{nom, s} \cdot \mu_{load} + q_{load}$.
                $$T = \sum_{s} (L_s \cdot R_{tot, s}) \cdot \Omega_m$$
                $$T_{eff} = \sum_{s} (L_s \cdot R_{eff, s}) \cdot \Omega_m$$
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Tonnellaggio Ponderato ($T_w, T_{w,eff}$)}
                Include l'efficienza articolare $c_{load}$:
                $$T_w = \sum_{s} (L_s R_{tot, s} c_{load}) \Omega_m$$
                $$T_{w,eff} = \sum_{s} (L_s R_{eff, s} c_{load}) \Omega_m$$
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{3.4 TUT (Time Under Tension) e Decadimento di Velocità}
    \begin{itemize}
        \item Tempo base per ripetizione: $\tau_0 = t_{conc} + t_{iso,short} + t_{ecc} + t_{iso,leng}$.
        \item Indice di sforzo istantaneo per la rip $j$: $\Delta_{RPE}(j) = RPE - (R_{base} - 1 - j)$.
    \end{itemize}
    \begin{equation*}
    \delta(j, RPE) = \begin{cases} 
    1.60\,\text{s} & \text{se } \Delta_{RPE} > 9.0 \quad (\text{grind al cedimento}) \\
    1.00\,\text{s} & \text{se } 8.0 < \Delta_{RPE} \le 9.0 \\
    0.60\,\text{s} & \text{se } 7.0 < \Delta_{RPE} \le 8.0 \\
    0.35\,\text{s} & \text{se } 6.0 < \Delta_{RPE} \le 7.0 \\
    0.15\,\text{s} & \text{se } 5.0 < \Delta_{RPE} \le 6.0 \\
    0.00\,\text{s} & \text{se } \Delta_{RPE} \le 5.0 
    \end{cases}
    \end{equation*}
    $$\text{TUT}_{set} = \sum_{j=0}^{R_{base}-1} \big(\tau_0 + \delta(j, RPE)\big) + R_{ass}\tau_0 + R_{part}(0.5\tau_0)$$
\end{frame}

\begin{frame}{3.5-3.6 Sets ($S$) e Modello Esponenziale di Fatica ($F$)}
    \begin{block}{Conteggio Serie}
        $$S = \sum_{s} 1 \cdot \Omega_m$$
    \end{block}
    \begin{block}{Modello di Fatica Sistemica/Tessutale Accumulata}
        $$F_{base\_rep}(j) = \big(\tau_0 + \delta(j, RPE)\big) \cdot (1.1)^{\max(0, \, \Delta_{RPE}(j) - 7.5)} \cdot f_{base} \cdot c_{load}$$
        \vspace{0.1cm}
        $$F_{set} = \sum_{j=0}^{R_{base}-1} F_{base\_rep}(j) + \underbrace{\left(R_{ass} \tau_0 1.1^{-0.5} f_{base} c_{load} c_{ass}\right)}_{\text{Assistite}} + \underbrace{\left(R_{part} 0.5\tau_0 f_{base} c_{load} \frac{c_{part}}{0.5}\right)}_{\text{Parziali}}$$
        $$F_{tot} = \sum_{s} F_{set, s} \cdot \Omega_m$$
    \end{block}
\end{frame}

\begin{frame}{3.7 Curve di Tensione Cumulative Muscolari ($V_m(t)$)}
    \begin{block}{Modellazione delle Parziali in Allungamento}
        Le ripetizioni parziali operano esclusivamente nel primo terzo del ROM ($t \in [0, 0.333]$):
        $$\chi_{part}(t) = \begin{cases} 1 & \text{se } t \in [0, \, 0.333] \\ 0 & \text{se } t \in (0.333, \, 1.0] \end{cases}$$
    \end{block}
    \begin{alertblock}{Curva Cumulativa di Tensione sul Distretto Muscolare $m$}
        $$V_m(t) = \sum_{e \in \text{Workout}} \Big( R_{base, e} \cdot \tilde{\mathbf{y}}_{e,m}(t) + R_{part, e} \cdot \tilde{\mathbf{y}}_{e,m}(t) \cdot \chi_{part}(t) \Big)$$
    \end{alertblock}
    \begin{itemize}
        \item Spiega l'accumulo asimmetrico e l'ipertrofia mediata dallo stretch prolungato.
    \end{itemize}
\end{frame}

\begin{frame}{4.1 Formulazione del Problema di Minimo Discreto Vincolato}
    \begin{itemize}
        \item \textbf{Spazio Combinatorio $\mathcal{S}$:}
        $$\mathbf{x} = \{ \text{giorni } d \in [1, D], \, \text{esercizi } e \in [N_{min}, N_{max}], \, \text{serie } s \in [S_{min}, S_{max}], \, \text{reps } r, \, \text{RPE} \}$$
    \end{itemize}
    \begin{block}{Funzione Obiettivo Multitermine}
        $$f(\mathbf{x}) = 10000 \cdot \sum_{i \in \text{Metriche}} w_i \frac{E_i(\mathbf{x})}{\sigma_i} + 1000 \cdot P_{penalties}(\mathbf{x})$$
    \end{block}
    \begin{itemize}
        \item $E_{curva}$: Errore di profilo di tensione ($L_2$ norm discreta contro target).
        \item $E_{vol}, E_{ton}, E_{tut}$: Scarti quadratici dai rapporti di efficacia ($r^* = 0.70$).
        \item $E_{distr}$: Divergenza tra volumi effettivi e distribuzioni target.
        \item $E_{variety}, E_{balance}$: Penalità di ridondanza e sbilanciamento giornaliero.
    \end{itemize}
\end{frame}

\begin{frame}{4.2 Calibrazione Monte Carlo dei Coefficienti $\sigma_i$}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \begin{block}{Problema della Scala Eterogenea}
                I termini d'errore $E_i$ hanno unità e ordini di grandezza disparati ($10^{-3}$ vs $10^4$).
            \end{block}
            \begin{alertblock}{Adimensionalizzazione Stocastica}
                Campionamento Monte Carlo preliminare su $K = 500$ configurazioni uniformi casuali:
                $$\sigma_i = 2 \cdot \mathbb{E}[E_i]$$
            \end{alertblock}
        \end{column}
        \begin{column}{0.42\textwidth}
            \textbf{Vantaggi Fondamentali:}
            \begin{itemize}
                \item Tutti i termini scalano in $[0, 1]$.
                \item I pesi $w_i$ assumono un reale significato intuitivo e comparabile.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{4.3 Algoritmo di Simulated Annealing (Ricottura Simulata)}
    \begin{block}{Criterio di Accettazione di Metropolis}
        A ogni iterazione $k$, dato $\mathbf{x}' = \text{Mutate}(\mathbf{x})$ e variazione $\Delta f = f(\mathbf{x}') - f(\mathbf{x})$:
        $$p(\mathbf{x} \to \mathbf{x}') = \begin{cases} 
        1 & \text{se } \Delta f < 0 \quad (\text{miglioramento deterministico}) \\ 
        \exp\left(-\frac{\Delta f}{T_k}\right) & \text{se } \Delta f \ge 0 \quad (\text{fuga stocastica da minimi locali})
        \end{cases}$$
    \end{block}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Raffreddamento Geometrico:}
            $$T_{k+1} = \alpha \cdot T_k, \quad \alpha = \left(\frac{T_{fin}}{T_0}\right)^{\frac{1}{N_{iter}}}$$
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Iperparametri:}
            \begin{itemize}
                \item $T_0 = 50.0$, $T_{fin} = 0.01$
                \item $N_{iter} \in [5000, 10000]$
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
