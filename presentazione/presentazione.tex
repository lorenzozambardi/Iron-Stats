% METODOLOGIA 1: LAGRANGE
% -------------------------------------------------------------
\begin{frame}{2.2 Metodo 1: Interpolazione Polinomiale di Lagrange (Teoria)}
    \begin{block}{Formulazione Globale di Lagrange}
        Dati $n$ punti campionati $(t_i, y_i)$ lungo il ROM $t \in [0, 1]$:
        $$p_{n-1}(t) = \sum_{i=1}^n y_i \ell_i(t), \qquad \ell_i(t) = \prod_{j \ne i} \frac{t - t_j}{t_i - t_j}$$
    \end{block}
    \begin{alertblock}{Condizionamento e Fenomeno di Runge}
        \begin{itemize}
            \item \textbf{Costante di Lebesgue:} $\Lambda_n = \max_{t \in [0, 1]} \sum_{i=1}^n |\ell_i(t)|$. Pur evitando la matrice di Vandermonde ($V \mathbf{a} = \mathbf{y}$), sui nodi equispaziati $\Lambda_n \sim \mathcal{O}(2^n)$: l'errore cresce esponenzialmente.
            \item \textbf{Fenomeno di Runge (Nodi Equispaziati):}
            $$\lim_{n \to \infty} \max_{t \in [0, 1]} |f(t) - p_{n-1}(t)| = \infty$$
            L'errore diverge esponenzialmente in corrispondenza degli estremi ($t \to 0$ e $t \to 1$).
        \end{itemize}
    \end{alertblock}
\end{frame}

\begin{frame}{2.2 Metodo 1: Lagrange su Nodi Equispaziati (Caso LR)}
    \centering
    \includegraphics[height=0.54\textheight, keepaspectratio]{plot_lagrange_runge.png}
    \begin{exampleblock}{Criticità Biomeccanica Evidenziata}
        \footnotesize Con soli $n=9$ nodi equispaziati, l'interpolante diverge drasticamente nella fase di allungamento iniziale ($t < 0.2$) e di accorciamento finale ($t > 0.85$), generando picchi di tensione artificiali e del tutto non fisiologici.
    \end{exampleblock}
\end{frame}

\begin{frame}{2.2 Metodo 1: Nodi Non Equispaziati di Gauss-Lobatto (Teoria)}
    \begin{block}{Distribuzione di Chebyshev-Gauss-Lobatto (CGL)}
        Per superare il fenomeno di Runge, si addensano i nodi agli estremi del ROM sull'intervallo $[0, 1]$:
        $$t_k = \frac{1}{2}\left(1 - \cos\frac{k\pi}{N-1}\right), \qquad k = 0, 1, \dots, N-1$$
    \end{block}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Vantaggio Teorico (Costante di Lebesgue)}
                L'errore di interpolazione soddisfa $\|f - p_{N-1}\|_\infty \le (1 + \Lambda_N) E_{N-1}(f)$:
                \begin{itemize}
                    \item Nodi equispaziati: $\Lambda_N \sim \mathcal{O}(2^N)$ (divergenza esponenziale).
                    \item \textbf{Nodi CGL:} $\Lambda_N \sim \frac{2}{\pi}\ln N + \mathcal{O}(1)$ (crescita solo logaritmica).
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{Perché Fallisce nella Biomeccanica?}
                \begin{itemize}
                    \item Servono \textbf{troppi parametri} ($N \ge 12$) per tentare di catturare la zona piatta di riposo ($t \in [0, 0.2]$).
                    \item La natura globale del polinomio genera comunque \textbf{ondulazioni parassite} e valori negativi non fisici ($y < 0$).
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

    \centering
    \includegraphics[height=0.54\textheight, keepaspectratio]{plot_lagrange_gauss_lobatto.png}
    \begin{exampleblock}{Analisi del Plot Comparativo su LR}

    \end{exampleblock}
\end{frame}

% METODOLOGIA 2: SPLINE CUBICHE
% -------------------------------------------------------------
\begin{frame}{2.2 Metodo 2: Spline Cubiche $C^2$ (Interpolazione a Tratti)}
    \begin{block}{Motivazione: Dal Globale al Locale}
        Per superare le oscillazioni dei polinomi globali (Runge), si suddivide l'intervallo $[0, 1]$ in $n$ sottointervalli $[t_k, t_{k+1}]$, definendo su ciascuno un polinomio cubico locale $S_k(t)$.
    \end{block}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Proprietà di Raccordo $C^2$}
                \footnotesize
                \begin{itemize}
                    \item \textbf{Interpolazione:} la curva passa esattamente per i nodi: $S(t_k) = y_k$.
                    \item \textbf{Continuità $C^2$:} derivata prima e seconda continue nei nodi interni per garantire massima regolarità.
                    \item \textbf{Bordo Naturale:} curvatura nulla agli estremi liberi, $S''(t_0) = S''(t_n) = 0$.
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Vantaggi Teorici}
                \footnotesize
                \begin{itemize}
                    \item \textbf{Assenza di Runge:} l'errore resta confinato localmente e non diverge agli estremi.
                    \item \textbf{Minima curvatura:} tra tutte le funzioni $C^2$ interpolanti, minimizza l'energia di deformazione:
                    $$\int_0^1 [S''(t)]^2 \, dt$$
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2.2 Metodo 2: Criticità Biomeccanica delle Spline (Undershoot)}
    \centering
    \includegraphics[height=0.48\textheight, keepaspectratio]{plot_spline_overshoot.png}
    \begin{alertblock}{Mancata Conservazione della Positività}
        \footnotesize
        \begin{itemize}
            \item \textbf{Undershoot non fisico ($y(t) < 0$):} la rigidità globale della continuità $C^2$ genera oscillazioni negative nella transizione da zona a riposo ad attiva ($t \in [0, 0.2]$). In fisiologia muscolare non esiste tensione attiva repulsiva.
            \item \textbf{Il dilemma applicativo:} per contenere le oscillazioni occorrerebbero nodi fittissimi ($N \ge 10$), rendendo il modello e l'interfaccia utente ingestibili.
        \end{itemize}
    \end{alertblock}
\end{frame}

\begin{frame}{2.2 Metodo 3: Curve Parametriche di Bézier (Formulazione 2D)}
    \begin{block}{Definizione Vettoriale e per Componenti}
        Ogni punto di controllo è un vettore nel piano ROM-Tensione: $\mathbf{P}_i = \begin{pmatrix} x_i \\ y_i \end{pmatrix} \in \mathbb{R}^2$. \\
        La curva è una combinazione lineare pesata a due coordinate:
        $$\mathbf{P}(t) = \begin{pmatrix} x(t) \\ y(t) \end{pmatrix} = \sum_{i=0}^3 \begin{pmatrix} x_i \\ y_i \end{pmatrix} B_{i,3}(t), \quad t \in [0, 1]$$
        $$\begin{cases}
        x(t) = B_{0,3}(t) x_0 + B_{1,3}(t) x_1 + B_{2,3}(t) x_2 + B_{3,3}(t) x_3 \\
        y(t) = B_{0,3}(t) y_0 + B_{1,3}(t) y_1 + B_{2,3}(t) y_2 + B_{3,3}(t) y_3
        \end{cases}$$
    \end{block}
    \begin{exampleblock}{Cosa Sono i Coefficienti $B_{i,3}(t)$?}
        \footnotesize Sono le \textbf{funzioni di miscelazione (blending functions)} di Bernstein: ad ogni istante $t \in [0, 1]$, stabiliscono il peso percentuale con cui ciascun punto $\mathbf{P}_i$ attira la curva a sé.
    \end{exampleblock}
\end{frame}

\begin{frame}{2.2 Metodo 3: I Polinomi Base di Bernstein ($B_{i,3}$)}
    \begin{block}{Definizione Analitica Generale}
        I polinomi di Bernstein di grado $n=3$ sono definiti dal binomio:
        $$B_{i,3}(t) = \binom{3}{i} t^i (1-t)^{3-i}, \quad \text{con } \binom{3}{i} = \frac{3!}{i!(3-i)!}, \quad t \in [0, 1]$$
    \end{block}
    \vspace{0.1cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Espressioni Esplicite ($n=3$)}
                \begin{itemize}
                    \item $\mathbf{B_{0,3}(t) = (1-t)^3}$
                    \item $\mathbf{B_{1,3}(t) = 3t(1-t)^2}$
                    \item $\mathbf{B_{2,3}(t) = 3t^2(1-t)}$
                    \item $\mathbf{B_{3,3}(t) = t^3}$
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.50\textwidth}
            \begin{exampleblock}{Dinamica di Ponderazione}
                \footnotesize
                \begin{itemize}
                    \item A $t=0$: $B_{0,3}=1$, gli altri $0 \implies \mathbf{P}(0) = \mathbf{P}_0$.
                    \item A $t=1/3$: picco di influenza per $\mathbf{P}_1$.
                    \item A $t=2/3$: picco di influenza per $\mathbf{P}_2$.
                    \item A $t=1$: $B_{3,3}=1$, gli altri $0 \implies \mathbf{P}(1) = \mathbf{P}_3$.
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2.2 Metodo 3: Proprietà dei Bernstein (Partizione Unità \& Positività)}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{1. Partizione dell'Unità}
                Dallo sviluppo del binomio di Newton:
                $$\sum_{i=0}^3 B_{i,3}(t) = \big((1-t) + t\big)^3 \equiv 1$$
                \footnotesize I pesi sommano sempre al $100\%$.
                \vspace{0.1cm} \\
                \textbf{Invarianza Affine:} cambi di scala (es. da gradi a $[0, 1]$ o carichi in kg) si applicano solo ai 4 punti $\mathbf{P}_i$ senza dover ricalcolare la curva.
            \end{exampleblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{2. Positività Rigorosa}
                Poiché $t \ge 0$ e $(1-t) \ge 0$:
                $$B_{i,3}(t) \ge 0 \implies \mathbf{P}(t) \in \text{Conv}(\{\mathbf{P}_i\})$$
                \footnotesize \textbf{Garanzia fisica fondamentale:}
                $$y_i \ge 0 \;\; \forall i \implies y(t) \ge 0 \quad \forall t$$
                La tensione muscolare non può mai scendere sotto zero per costruzione geometrica.
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2.2 Metodo 3: Cosa Significa Matematicamente "Attirare la Curva"?}
    \begin{columns}[T]
        \begin{column}{0.49\textwidth}
            \begin{block}{1. Centro di Massa a Masse Variabili}
                La curva è formalmente identica al \textbf{baricentro} di 4 masse collocate nei nodi:
                $$\mathbf{P}(t) = \frac{\sum_{i=0}^3 m_i(t) \mathbf{P}_i}{\sum_{i=0}^3 m_i(t)}, \quad \text{con } m_i(t) = B_{i,3}(t)$$
                \footnotesize
                \begin{itemize}
                    \item A $t=1/3$, la massa $B_{1,3}(1/3) \approx 0.44$ è massima: il centro di massa viene \textbf{trascinato fisicamente verso $\mathbf{P}_1$}.
                    \item Non tocca mai $\mathbf{P}_1$ perché le altre masse $B_0, B_2 > 0$ esercitano una contro-trazione.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.49\textwidth}
            \begin{exampleblock}{2. Vettori Tangente e Pendenza}
                Derivando analiticamente la curva $\mathbf{P}'(t) = \frac{d\mathbf{P}}{dt}$:
                $$\mathbf{P}'(0) = 3(\mathbf{P}_1 - \mathbf{P}_0), \quad \mathbf{P}'(1) = 3(\mathbf{P}_3 - \mathbf{P}_2)$$
                \footnotesize
                \begin{itemize}
                    \item $\mathbf{P}_1$ impone la \textbf{direzione di lancio}: il vettore tangente iniziale punta esattamente verso $\mathbf{P}_1$.
                    \item $\mathbf{P}_2$ impone la \textbf{direzione di atterraggio} in $\mathbf{P}_3$.
                \end{itemize}
            \end{exampleblock}
            \begin{alertblock}{Approssimazione vs Interpolazione}
                \footnotesize Non dover attraversare $\mathbf{P}_1$ e $\mathbf{P}_2$ evita curvature forzate ed elimina Runge alla radice.
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2.2 Metodo 3: Costruzione Geometrica (Algoritmo di De Casteljau)}
    \begin{block}{Il Principio dei Tagli Proporzionali Progressivi}
        Dato $t \in [0, 1]$, la curva si costruisce dividendo ricorsivamente i segmenti in rapporto $t : (1-t)$:
        \begin{itemize}\setlength{\itemsep}{1pt}
            \item \textbf{Passo 1 (Poligono di Controllo):} sui 3 segmenti $\{P_i P_{i+1}\}$, individua 3 punti intermedi:
            $$Q_i(t) = (1-t)P_i + t P_{i+1}, \quad i \in \{0, 1, 2\}$$
            \item \textbf{Passo 2 (Segmenti Intermedi):} collegando i $\{Q_i\}$, individua 2 punti:
            $$R_i(t) = (1-t)Q_i(t) + t Q_{i+1}(t), \quad i \in \{0, 1\}$$
            \item \textbf{Passo 3 (Punto Finale):} sul segmento $R_0 - R_1$, individua il punto sulla curva:
            $$\mathbf{P}(t) = (1-t)R_0(t) + t R_1(t)$$
        \end{itemize}
    \end{block}
    \begin{exampleblock}{Proprietà Notevole della Tangente Istantanea}
        \footnotesize Il segmento $R_0 - R_1$ non trova solo $\mathbf{P}(t)$: \textbf{è esattamente la retta tangente alla curva in quel punto.}
    \end{exampleblock}
\end{frame}

\begin{frame}{2.2 Metodo 3: Visualizzazione Geometrica su LR ($t = 0.5$)}
    \centering
    \includegraphics[height=0.52\textheight, keepaspectratio]{plot_de_casteljau.png}
    \begin{block}{Lettura Grafica della Costruzione Geometrica su LR}
        \footnotesize
        \begin{itemize}
            \item \textbf{Tratti tratteggiati grigi:} poligono di controllo originario $\{P_0, P_1, P_2, P_3\}$.
            \item \textbf{Segmenti blu e punti $Q_i$:} punti medi ($t=0.5$) dei segmenti di controllo iniziali.
            \item \textbf{Segmento arancione $R_0 - R_1$:} unione dei punti di secondo livello; è tangente alla curva in $\mathbf{P}(0.5)$.
            \item \textbf{Punto rosso $\mathbf{P}(0.5)$:} punto medio di $R_0 - R_1$, giacente esattamente sulla traiettoria biomeccanica.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{2.2 Metodo 3: Proprietà Geometrica di Suddivisione (Subdivision)}
    \begin{block}{Teorema della Suddivisione Ricorsiva}
        La costruzione di De Casteljau non si limita a valutare $\mathbf{P}(t)$: genera automaticamente i punti di controllo per \textbf{spezzare la curva in due sub-curve di Bézier esatte.}
    \end{block}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Ramo Sinistro (Intervallo $[0, t]$)}
                Curva di Bézier cubica con nodi di controllo:
                $$\mathcal{C}_{\text{sinistra}} = \{P_0, \; Q_0, \; R_0, \; \mathbf{P}(t)\}$$
            \end{exampleblock}
            \vspace{0.1cm}
            \begin{alertblock}{Ramo Destro (Intervallo $[t, 1]$)}
                Curva di Bézier cubica con nodi di controllo:
                $$\mathcal{C}_{\text{destra}} = \{\mathbf{P}(t), \; R_1, \; Q_2, \; P_3\}$$
            \end{alertblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Applicazioni Computazionali}
                \footnotesize
                \begin{itemize}
                    \item \textbf{Raffinamento Locale:} calcolo della quadratura e dell'area sottesa per integrazione numerica adattiva senza valutare polinomi complessi.
                    \item \textbf{Zero Divisioni:} solo somme pesate $a(1-t) + bt$, stabili al $100\%$ anche in aritmetica a precisione ridotta.
                    \item \textbf{Complessità $\mathcal{O}(1)$:} 6 semplici interpolazioni lineari 2D.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2.2 Metodo 3: Come si Calcola la Tensione $y(x)$ nel Software}
    \begin{block}{Il Problema dell'Inversione Cinematica}
        Bézier è parametrizzata in $t \in [0, 1] \implies \begin{cases} x(t) = (1-t)^3 x_0 + 3t(1-t)^2 x_1 + 3t^2(1-t) x_2 + t^3 x_3 \\ y(t) = (1-t)^3 y_0 + 3t(1-t)^2 y_1 + 3t^2(1-t) y_2 + t^3 y_3 \end{cases}$ \\
        Per calcolare la tensione $y$ dato un angolo di ROM target $x_{target} \in [0, 1]$:
    \end{block}
    \vspace{0.1cm}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \begin{alertblock}{Risoluzione Radice $x(t) - x_{target} = 0$}
                \begin{itemize}
                    \item Poiché $x_0 < x_1 < x_2 < x_3$, la funzione $x(t)$ è \textbf{strettamente monotona crescente}.
                    \item Esiste un'\textbf{unica soluzione} $t^* \in [0, 1]$.
                    \item Trovata con \textbf{bisezione} in max 15 iterazioni:
                    $$|x(t^{(k)}) - x_{target}| < 10^{-3}$$
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Valutazione Diretta di $y$}
                Una volta calcolato $t^*$, la tensione vale:
                $$y_{target} = y(t^*)$$
                \begin{itemize}
                    \item Se $x_{target} < x_0$ o $x_{target} > x_3 \implies y = 0$.
                    \item Costo costante $\mathcal{O}(1)$ in frazioni di microsecondo.
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

    \begin{exampleblock}{Proprietà Emergenti della Curva di Bézier}
        \footnotesize \textbf{Parsimonia estrema:} soli 4 nodi di controllo $P_0, P_1, P_2, P_3$ descrivono l'intero profilo biomeccanico. \\
        \textbf{Complessità $\mathcal{O}(1)$:} calcolabile direttamente o tramite schema stabile di De Casteljau, ideale per valutazioni ripetute nell'ottimizzatore.
    \end{exampleblock}
\end{frame}

\begin{frame}{2.3 Quadratura Numerica e Riscalamento Vincolato}
    \begin{block}{Calcolo dell'Area Sotto la Curva (AUC)}
        \footnotesize
        Griglia discreta $N = 50$, passo $h = \Delta t = \frac{1}{N-1}$:
        $$I_{trap} = \int_0^1 y(t)\,dt \approx \sum_{k=0}^{N-2} \frac{y(t_k) + y(t_{k+1})}{2} (t_{k+1} - t_k), \qquad E_{trap} = -\frac{h^2}{12} y''(\xi) = \mathcal{O}(h^2)$$
    \end{block}
    \begin{block}{Operatore di Riscalamento Funzionale}
        \footnotesize
        Normalizza la curva rispetto alla quadratura discreta calcolata:
        $$\tilde{y}(t) = \begin{cases} y(t) \cdot \frac{M}{I_{trap}} & \text{se } I_{trap} > 0 \\ 0 & \text{altrimenti} \end{cases}$$
        \begin{itemize}\setlength{\itemsep}{1pt}
            \item \textbf{Consistenza discreta (software):} $I_{trap}(\tilde{y}) \equiv M$ (esatta sulla griglia di calcolo).
            \item \textbf{Limite continuo:} $\int_0^1 \tilde{y}(t)\,dt = M + \mathcal{O}(h^2)$ (a meno del troncamento $\mathcal{O}(h^2)$).
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}

    \vspace{0.4cm}
    {\normalsize \textcolor{secondaryColor}{Volume, Tonnellaggio, TUT e Fatica}}

\end{frame}

\begin{frame}{3.1 Volume ($V$) ed Effective Volume ($V_{\mathrm{eff}}$)}
    \begin{itemize}
        \item \textbf{Ripetizioni Totali Equivalenti (serie $s$):}
        $$R_{\mathrm{tot}, s} = R_{\mathrm{base}, s} + c_{\mathrm{ass}} R_{\mathrm{ass}, s} + c_{\mathrm{part}} R_{\mathrm{part}, s} \quad (c_{\mathrm{ass}}=0.5, \, c_{\mathrm{part}}=0.33)$$
        \item \textbf{Ripetizioni Efficaci da Scala RPE ($RPE \in [0, 10]$):}
        $$R_{\mathrm{eff}, s} = \min\Big(R_{\mathrm{base}, s}, \, \max\big(0, \, RPE_s - 4.0\big)\Big) + c_{\mathrm{ass}} R_{\mathrm{ass}, s} + c_{\mathrm{part}} R_{\mathrm{part}, s}$$
        \item \textbf{Proiezione sulla quota muscolare $\Omega_{e(s), m} = \int_0^1 \tilde{y}_{e(s), m}(t)\,dt$:}
        $$V = \sum_{s} R_{\mathrm{tot}, s} \cdot \Omega_{e(s), m}, \qquad V_{\mathrm{eff}} = \sum_{s} R_{\mathrm{eff}, s} \cdot \Omega_{e(s), m}$$
    \end{itemize}
\end{frame}

\begin{frame}{3.2-3.3 Tonnellaggio e Tonnellaggio Ponderato}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Tonnellaggio ($T$ e $T_{\mathrm{eff}}$)}
                Carico effettivo: $L_s = L_{\mathrm{nom}, s} \cdot \mu_{\mathrm{load}} + q_{\mathrm{load}}$.
                $$T = \sum_{s} (L_s \cdot R_{\mathrm{tot}, s}) \cdot \Omega_{e(s), m}$$
                $$T_{\mathrm{eff}} = \sum_{s} (L_s \cdot R_{\mathrm{eff}, s}) \cdot \Omega_{e(s), m}$$
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Tonnellaggio Ponderato ($T_w, T_{w,\mathrm{eff}}$)}
                Include l'efficienza articolare $c_{\mathrm{load}, s}$:
                $$T_w = \sum_{s} (L_s R_{\mathrm{tot}, s} c_{\mathrm{load}, s}) \Omega_{e(s), m}$$
                $$T_{w,\mathrm{eff}} = \sum_{s} (L_s R_{\mathrm{eff}, s} c_{\mathrm{load}, s}) \Omega_{e(s), m}$$
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{3.4 TUT (Time Under Tension) e Decadimento di Velocità}
    \footnotesize
    \begin{itemize}
        \item Tempo base per rip: $\tau_0 = t_{\mathrm{conc}} + t_{\mathrm{iso,short}} + t_{\mathrm{ecc}} + t_{\mathrm{iso,leng}}$; sforzo: $\Delta_{RPE}(j) = RPE - (R_{\mathrm{base}} - 1 - j)$.
    \end{itemize}
    \vspace{-0.15cm}
    \begin{equation*}
    \delta(j, RPE) = \begin{cases} 
    1.60\,\text{s} & \text{se } \Delta_{RPE} > 9.0 \\
    1.00\,\text{s} & \text{se } 8.0 < \Delta_{RPE} \le 9.0 \\
    0.60\,\text{s} & \text{se } 7.0 < \Delta_{RPE} \le 8.0 \\
    0.35\,\text{s} & \text{se } 6.0 < \Delta_{RPE} \le 7.0 \\
    0.15\,\text{s} & \text{se } 5.0 < \Delta_{RPE} \le 6.0 \\
    0.00\,\text{s} & \text{se } \Delta_{RPE} \le 5.0 
    \end{cases}
    \end{equation*}
    \vspace{-0.2cm}
    $$\text{TUT}_{\mathrm{set}} = \sum_{j=0}^{R_{\mathrm{base}}-1} \big(\tau_0 + \delta(j, RPE)\big) + R_{\mathrm{ass}}\tau_0 + R_{\mathrm{part}}(0.5\tau_0), \quad \text{TUT}_{\mathrm{tot}} = \sum_{s} \text{TUT}_{\mathrm{set}, s} \cdot \Omega_{e(s), m}$$
    \begin{block}{TUT Efficace ($\text{TUT}_{\mathrm{eff}}$)}
        \footnotesize
        Somma ristretta alle sole ultime $k = \operatorname{round}(R_{\mathrm{eff,base}})$ ripetizioni della serie (più assistite e parziali):
        $$\text{TUT}_{\mathrm{eff}} = \sum_s \left( \sum_{j=R_{\mathrm{base}}-k}^{R_{\mathrm{base}}-1} \big(\tau_0 + \delta(j, RPE)\big) + R_{\mathrm{ass}}\tau_0 + R_{\mathrm{part}}(0.5\tau_0) \right) \cdot \Omega_{e(s), m}$$
    \end{block}
\end{frame}

\begin{frame}{3.5-3.6 Sets ($S$) e Modello Esponenziale di Fatica ($F$)}
    \begin{block}{Conteggio Serie}
        $$S = \sum_{s} 1 \cdot \Omega_{e(s), m}$$
    \end{block}
    \begin{block}{Modello di Fatica Sistemica/Tessutale Accumulata}
        $$F_{\mathrm{base\_rep}}(j) = \big(\tau_0 + \delta(j, RPE)\big) \cdot (1.1)^{\max(0, \, \Delta_{RPE}(j) - 7.5)} \cdot f_{\mathrm{base}} \cdot c_{\mathrm{load}}$$
        \vspace{0.1cm}
        $$F_{\mathrm{set}} = \sum_{j=0}^{R_{\mathrm{base}}-1} F_{\mathrm{base\_rep}}(j) + \underbrace{R_{\mathrm{ass}} \tau_0 1.1^{-0.5} f_{\mathrm{base}} c_{\mathrm{load}} c_{\mathrm{ass}}}_{\text{Assistite}} + \underbrace{R_{\mathrm{part}} 0.5\tau_0 f_{\mathrm{base}} c_{\mathrm{load}} \frac{c_{\mathrm{part}}}{0.5}}_{\text{Parziali}}$$
        $$F_{\mathrm{tot}} = \sum_{s} F_{\mathrm{set}, s} \cdot \Omega_{e(s), m}$$
    \end{block}
\end{frame}

\begin{frame}{3.7 Curva di Tensione Cumulativa Muscolare ($V_m(t)$)}
    \begin{block}{Modellazione delle Parziali in Allungamento}
        Le ripetizioni parziali operano esclusivamente nel primo terzo del ROM ($t \in [0, 0.333]$):
        $$\chi_{\mathrm{part}}(t) = \begin{cases} 1 & \text{se } t \in [0, \, 0.333] \\ 0 & \text{se } t \in (0.333, \, 1.0] \end{cases}$$
    \end{block}
    \begin{alertblock}{Curva Cumulativa di Tensione sul Distretto Muscolare $m$}
        $$V_m(t) = \sum_{e \in \text{Workout}} v_e \cdot \Big( (R_{\mathrm{base}, e} + c_{\mathrm{ass}} R_{\mathrm{ass}, e}) \cdot \tilde{\mathbf{y}}_{e,m}(t) + R_{\mathrm{part}, e} \cdot \tilde{\mathbf{y}}_{e,m}(t) \cdot \chi_{\mathrm{part}}(t) \Big)$$
    \end{alertblock}
    \begin{itemize}
        \item Spiega l'accumulo asimmetrico e l'ipertrofia mediata dallo stretch prolungato.
    \end{itemize}
\end{frame}

\begin{frame}{4.1 Introduzione: Limiti Scientifici e Bias nei Modelli}
    \begin{block}{Ipertrofia Muscolare: Un Territorio Scientifico Aperto}
        \footnotesize
        \begin{itemize}
            \item I meccanismi biologici che regolano l'ipertrofia (tensione meccanica, danno muscolare, stretch-mediated hypertrophy) \textbf{non sono ancora del tutto compresi} né descritti da leggi universali di primo principio.
            \item La letteratura scientifica presenta teorie in continua evoluzione e una marcata variabilità soggettiva nella risposta biomeccanica allo stimolo.
        \end{itemize}
    \end{block}
    \vspace{-0.1cm}
    \begin{alertblock}{L'Inevitabilità del Bias e il Problema degli Iperparametri}
        \footnotesize
        \begin{itemize}
            \item Qualsiasi formalizzazione matematica impone assunzioni a priori, introducendo un \textbf{bias teorico ed empirico inevitabilmente molto grande}.
            \item Lasciare tutti i pesi e vincoli aperti all'utente \textbf{non risolve il problema}: lo sposta interamente sulla \textbf{scelta arbitraria degli iperparametri}.
        \end{itemize}
    \end{alertblock}
    \vspace{-0.1cm}
    \begin{exampleblock}{Assenza di Dataset: Il Ruolo della Definizione Formale}
        \footnotesize
        Non esistono in letteratura dati quantitativi sufficienti per addestrare o calibrare empiricamente il modello: di questo solver \textbf{non resta altro che la rigorosa formulazione matematica del problema di ottimizzazione vincolato}.
    \end{exampleblock}
\end{frame}

\begin{frame}{4.2 Variabili Decisionali e Limiti Operativi di Default}
    \begin{block}{Variabili Decisionali Discrete}
        \footnotesize
        \begin{itemize}
            \item $x_{i,d} \in \{0, 1\}$: \textbf{Attivazione.} Vale $1$ se l'esercizio $i$ è programmato nel giorno $d$, $0$ altrimenti.
            \item $v_{i,d} \in \mathbb{N}$: \textbf{Volume} (numero di serie) per l'esercizio $i$ nel giorno $d$.
            \item $r_{i,d} \in \mathbb{N}$: \textbf{Ripetizioni} (reps per serie) dell'esercizio $i$ nel giorno $d$.
        \end{itemize}
    \end{block}
    \vspace{-0.1cm}
    \begin{exampleblock}{Limiti Operativi di Default (Nessun Vincolo Hard-Coded)}
        \footnotesize
        Dato l'alto bias intrinseco, \textbf{nessun vincolo è hard-coded}: l'utente ha piena libertà di scegliere vincoli estremamente rigidi o rilassarli da interfaccia. I valori di default correnti sono:
        \begin{itemize}
            \item \textbf{Microciclo e Densità:} $D \in [3, 7]$ giorni; esercizi per sessione $E_{\min} = 3, E_{\max} = 10$.
            \item \textbf{Volume e Reps:} serie $V_{\min} = 1, V_{\max} = 6$; reps $R_{\min} = 5, R_{\max} = 20$.
            \item \textbf{Fatica e Recupero:} limite fatica max/giorno $F_{\max} = 1500$; recuperi $80\,\text{s} \div 210\,\text{s}$.
            \item \textbf{Curve Target $T_m(t)$:} selezionabili tra Costante ($c$), Sigmoide, Sigmoide Inversa, Lineare.
        \end{itemize}
    \end{exampleblock}
\end{frame}

\begin{frame}{4.3 Funzione Obiettivo: Ottimizzazione Multi-Obiettivo Scalarizzata}
    \begin{block}{Pesi e Target di Efficienza (Impostabili da UI)}
        \footnotesize
        L'utente definisce i \textbf{Pesi ($w_k$)} (la cui somma è normalizzata a $1.0$ via Softmax) e i \textbf{Target Ideali}:
        \begin{itemize}
            \item $w_{\mathrm{curva}}$: peso per l'aderenza alla Curva Target di tensione muscolare.
            \item $w_{\mathrm{vol}}, w_{\mathrm{ton}}, w_{\mathrm{tut}}$: pesi per l'efficienza di Volume ($\mathrm{Target}_{\mathrm{vol}}$), Tonnellaggio ($\mathrm{Target}_{\mathrm{ton}}$) e TUT ($\mathrm{Target}_{\mathrm{tut}}$).
            \item $w_{\mathrm{distr}}$: peso per l'aderenza alla distribuzione volumetrica Macro/Micro richiesta.
            \item $w_{\mathrm{variety}}$: peso per incentivare la diversità degli esercizi su tutto il microciclo.
            \item $w_{\mathrm{balance}}$: peso per il bilanciamento di volume e fatica tra i giorni.
        \end{itemize}
    \end{block}
    \vspace{-0.1cm}
    \begin{alertblock}{Principio di Uniformazione nel Dominio $[0.0, 1.0]$}
        \footnotesize
        Per garantire un impatto simmetrico ai pesi $w$, \textbf{tutti gli errori sono matematicamente normalizzati in $[0, 1]$} dividendo per i massimi teorici o mediando gli scarti.
    \end{alertblock}
    \vspace{-0.1cm}
    \begin{exampleblock}{Problema di Minimo Globale}
        $$\min \Big( w_{\mathrm{curva}} E_{\mathrm{curva}} + w_{\mathrm{vol}} E_{\mathrm{vol}} + w_{\mathrm{ton}} E_{\mathrm{ton}} + w_{\mathrm{tut}} E_{\mathrm{tut}} + w_{\mathrm{distr}} E_{\mathrm{distr}} + w_{\mathrm{variety}} E_{\mathrm{variety}} + w_{\mathrm{balance}} E_{\mathrm{balance}} \Big)$$
    \end{exampleblock}
\end{frame}

\begin{frame}{4.4 Calcolo degli Errori Normalizzati $[0, 1]$: Curva ed Efficienza}
    \begin{block}{Errore Curva (Media del MSE Normalizzato)}
        \footnotesize
        L'errore quadratico medio sui punti discreti del ROM viene normalizzato rispetto al picco della curva cumulativa per confinarlo rigorosamente in $[0, 1]$:
        $$E_{\mathrm{curva}} = \frac{1}{|M|} \sum_{m \in M} \frac{1}{|T|} \sum_{t \in T} \left( \frac{V_m(t)}{\max_{\tau \in T} V_m(\tau)} - T_m(t) \right)^2$$
    \end{block}
    \vspace{-0.1cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Errore Volume}
                \footnotesize
                $$E_{\mathrm{vol}} = \left( \frac{\text{Effective Volume}}{\text{Total Volume}} - \mathrm{Target}_{\mathrm{vol}} \right)^2$$
            \end{exampleblock}
            \begin{exampleblock}{Errore Tonnellaggio}
                \footnotesize
                $$E_{\mathrm{ton}} = \left( \frac{\text{Effective Tonnage}}{\text{Total Tonnage}} - \mathrm{Target}_{\mathrm{ton}} \right)^2$$
            \end{exampleblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Errore TUT}
                \footnotesize
                $$E_{\mathrm{tut}} = \left( \frac{\text{Effective TUT}}{\text{Total TUT}} - \mathrm{Target}_{\mathrm{tut}} \right)^2$$
            \end{exampleblock}
            \begin{block}{Confinamento Rigoroso}
                \footnotesize
                Essendo frazioni di efficacia $\in [0, 1]$ e target $\in [0, 1]$, il quadrato dello scarto appartiene nativamente all'intervallo $[0.0, 1.0]$.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{4.5 Calcolo degli Errori Normalizzati $[0, 1]$: Distribuzione e Bilanciamento}
    \begin{block}{Errore Distribuzione Volumi (Distanza Massima Normalizzata)}
        \footnotesize
        Target anatomico globale: $\mathrm{Target}_{\mathrm{globale}}(m) = \mathrm{Macro\%}(M) \times \mathrm{Sub\%}(m)$. \\
        Normalizzato dividendo per $2.0$ (massima distanza quadratica teorica tra due distribuzioni percentuali):
        $$E_{\mathrm{distr}} = \frac{1}{2.0} \sum_{m} \left( \frac{V_{\mathrm{effettivo}}(m)}{V_{\mathrm{totale}}} - \mathrm{Target}_{\mathrm{globale}}(m) \right)^2$$
    \end{block}
    \vspace{-0.1cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Errore Varietà Esercizi}
                \footnotesize
                Incentiva la rotazione dei movimenti:
                $$E_{\mathrm{variety}} = \left( 1 - \frac{\text{Unique Exercises}}{\text{Total Exercises}} \right)^2$$
            \end{exampleblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Errore Bilanciamento Giornaliero}
                \footnotesize
                Varianza divisa per la varianza teorica massima:
                $$E_{\mathrm{balance}} = \frac{1}{2} \left( \frac{\operatorname{Var}(\text{Fatigue})}{\operatorname{Var}_{\max\_F}} + \frac{\operatorname{Var}(\text{Volume})}{\operatorname{Var}_{\max\_V}} \right)$$
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{4.6 Vincoli Fisiologici (Constraints)}
    \begin{block}{Limiti di Serie e Reps (Formulazione Big-M)}
        \footnotesize
        Se l'esercizio è scelto ($x_{i,d} = 1$), rispetta i limiti; se inattivo ($x_{i,d} = 0$), i valori collassano a 0:
        $$V_{\min} \cdot x_{i,d} \le v_{i,d} \le V_{\max} \cdot x_{i,d} \qquad (\text{default: } 1 \le v_{i,d} \le 6)$$
        $$R_{\min} \cdot x_{i,d} \le r_{i,d} \le R_{\max} \cdot x_{i,d} \qquad (\text{default: } 5 \le r_{i,d} \le 20)$$
    \end{block}
    \vspace{-0.2cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Densità Minima e Massima}
                \footnotesize
                $$E_{\min} \le \sum_{i \in \mathcal{E}} x_{i,d} \le E_{\max} \quad (3 \le \text{Esercizi/Sessione} \le 10)$$
            \end{exampleblock}
            \begin{alertblock}{Overlap Neurale (Unicità Esercizio)}
                \footnotesize
                Penalità al Costo per impedire lo stesso esercizio duplicato in slot distinti $s$ dello stesso giorno $d$:
                $$\text{Penalty se } \sum_{s \in \mathcal{S}_d} x_{s, i, d} > 1 \quad \forall i \in \mathcal{E}$$
            \end{alertblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Recupero Deterministico ($\mathrm{rest}_i$)}
                \footnotesize
                Calcolato a priori dalla fatica sistemica $F_i$:
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Bassa Fatica ($F_i \le 2.0$):} $60\,\text{s}$ fisso.
                    \item \textbf{Media Fatica ($2 < F_i \le 7$):} scala linearmente da $1'$ a $3'$.
                    \item \textbf{Alta Fatica ($F_i > 7.0$):} scala linearmente da $3'$ a $3'30''$ (es. Stacco).
                \end{itemize}
                Valori arrotondati al mezzo minuto più vicino.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
